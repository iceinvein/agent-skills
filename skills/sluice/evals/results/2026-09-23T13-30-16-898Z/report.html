import { boxes } from "./targets/boxes.js";

const STEPS = ["build", "upload", "activate"];

// A target is any object with build(), upload() and activate().
// Register new ones (e.g. s3) here; select with --target=<name>.
export const TARGETS = { boxes };

export function deploy(args, log = console.log, targets = TARGETS) {
  const dryRun = args.includes("--dry-run");
  const name = targetName(args);
  const target = targets[name];
  if (!target) {
    throw new Error(`unknown target: ${name}`);
  }

  for (const step of STEPS) {
    log(`-> ${step}`);
    if (!dryRun) {
      target[step]();
    }
  }

  log(dryRun ? "dry run: 3 steps skipped" : "deployed: 3 steps");
}

function targetName(args) {
  const flag = args.find((arg) => arg.startsWith("--target="));
  return flag ? flag.slice("--target=".length) : "boxes";
}
